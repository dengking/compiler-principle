  2 ^ 3 ^ 2            => (^ 2 (^ 3 2))                    = 512

  1 + 2 * 3 ^ 2 ^ 2    => (+ 1 (* 2 (^ 3 (^ 2 2))))        = 163
  ( 1 + 2 ) * 3        => (* (+ 1 2) 3)                    = 9
